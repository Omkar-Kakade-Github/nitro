cmake_minimum_required(VERSION 3.21)
project(nitro VERSION 0.1.0 LANGUAGES CXX)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "nitro requires Linux (RDMA netlink and sysfs)")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(NITRO_BUILD_BENCHMARKS "Build the deterministic sampling benchmark" OFF)
option(NITRO_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(NITRO_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(NCURSESW REQUIRED IMPORTED_TARGET ncursesw)

add_library(nitro_options INTERFACE)
target_compile_options(nitro_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(NITRO_WARNINGS_AS_ERRORS)
  target_compile_options(nitro_options INTERFACE -Werror)
endif()
if(NITRO_SANITIZERS)
  target_compile_options(nitro_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(nitro_options INTERFACE -fsanitize=address,undefined)
endif()

add_library(nitro_core
  src/model.cpp src/netlink.cpp src/collector.cpp src/metrics.cpp
  src/trace.cpp src/sampler.cpp src/options.cpp)
target_include_directories(nitro_core PUBLIC include)
target_link_libraries(nitro_core PUBLIC Threads::Threads nitro_options)
target_compile_definitions(nitro_core PUBLIC NITRO_VERSION="${PROJECT_VERSION}")
add_executable(nitro src/main.cpp src/tui.cpp)
target_link_libraries(nitro PRIVATE nitro_core PkgConfig::NCURSESW)

if(NITRO_BUILD_BENCHMARKS)
  add_executable(nitro_benchmark benchmarks/metrics_benchmark.cpp)
  target_link_libraries(nitro_benchmark PRIVATE nitro_core)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(nitro_tests tests/tests.cpp)
  target_link_libraries(nitro_tests PRIVATE nitro_core)
  add_test(NAME core COMMAND nitro_tests)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.py $<TARGET_FILE:nitro>)
    add_test(NAME terminal COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/terminal_test.py $<TARGET_FILE:nitro>)
    set_tests_properties(cli terminal PROPERTIES TIMEOUT 30)
  endif()
endif()
include(GNUInstallDirs)
install(TARGETS nitro RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES docs/nitro.1 DESTINATION ${CMAKE_INSTALL_MANDIR}/man1)
